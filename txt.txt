# Write a program to implement the,
# i. Water jug problem using BFS and DFS search strategies. Input # jug capacities
# and goal state from the user. Display the sequence of states and # actions to reach
# the goal.

from collections import deque

def moves(x,y,a,b):
    return [
        ((a,y), "Fill Jug 1"),
        ((x,b), "Fill Jug 2"),
        ((0,y), "Empty Jug 1"),
        ((x,0), "Empty Jug 2"),
        ((x - min(x,b-y) , y+min(x,b-y)), "Pour 1 to 2"),
        ((x+min(y,a-x) , y - min(y,a-x)) , "Pour 2 to 1")
    ]

def bfs(a,b,goal):
    q = deque([((0,0) , [])])
    visited = set()

    while q:
        (x,y), path = q.popleft()

        if x == goal or y == goal:
            for action , state in path:
                print(action , "->" , state)
            print("goal reached:", (x,y))
            return

        if (x,y) in visited:
            continue
        visited.add((x,y))

        for state, action in moves (x,y,a,b):
            if state not in visited:
                q.append((state , path + [(action,state)]))
    print ("none")



def dfs(a, b, goal):
    stack = [((0, 0), [])]
    visited = set()

    while stack:
        (x, y), path = stack.pop()

        if x == goal or y == goal:
            for action, state in path:
                print(action, "->", state)
            print("Goal reached:", (x, y))
            return

        if (x, y) in visited:
            continue

        visited.add((x, y))

        for state, action in moves(x, y, a, b):
            if state not in visited:
                stack.append((state, path + [(action, state)]))

    print("None")

# Tic-Tac-Toe game agent, the agent should play optimally using the minimax
# algorithm. Allow a human user to play against the AI agent. Display the result

def show():
    for i in range(0, 9, 3):
        print(board[i], "|", board[i+1], "|", board[i+2])

def winner(p):
    wins = [(0,1,2), (3,4,5), (6,7,8),
            (0,3,6), (1,4,7), (2,5,8),
            (0,4,8), (2,4,6)]
    return any(all(board[i] == p for i in w) for w in wins)

def minimax(ai):
    if winner("O"): return 1
    if winner("X"): return -1
    if " " not in board: return 0

    scores = []
    for i in range(9):
        if board[i] == " ":
            board[i] = ai
            scores.append(minimax("X" if ai == "O" else "O"))
            board[i] = " "

    return max(scores) if ai == "O" else min(scores)

board = [" "] * 9

while True:

 # Implement the A* search algorithm for solving the 8-puzzle problem.
# i. Use an appropriate heuristic function (e.g., misplaced tiles or Manhattan
#  distance).
# ii. Display the sequence of states leading to the goal state along with the cost.

import heapq

goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

def heuristic(s):
    total_sum = 0
    for i in range(9):
        if s[i] != 0:
            if s[i] != goal[i]:
                total_sum += 1
    return total_sum

def solve(start):
    q = [(heuristic(start), 0, start, [])]
    visited = set()

    while q:
        f, cost, s, path = heapq.heappop(q)

        if s in visited:
            continue
        visited.add(s)
        path = path + [s]

        if s == goal:
            for state in path:
                print(state[:3])
                print(state[3:6])
                print(state[6:])
                print()
            print("Total cost:", cost)
            return

        z = s.index(0)

        for d in [-3, 3, -1, 1]:
            n = z + d

            if 0 <= n < 9 and not (d == -1 and z % 3 == 0) and not (d == 1 and z % 3 == 2):
                new = list(s)
                new[z], new[n] = new[n], new[z]
                new = tuple(new)

                if new not in visited:
                    g = cost + 1
                    heapq.heappush(q, (g + heuristic(new), g, new, path))

start = tuple(map(int, input("Enter 9 numbers (use 0 for blank): ").split()))
solve(start)

import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import MinMaxScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.metrics import silhouette_score

data = load_iris()
X = data.data
X = MinMaxScaler().fit_transform(X)

kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
k = kmeans.fit_predict(X)

plt.scatter(X[:, 0], X[:, 1], c=k, cmap='viridis')
plt.title("K-Means Clustering")
plt.xlabel("Feature 1")
plt.ylabel("Feature 2")
plt.show()

hierarchy = AgglomerativeClustering(n_clusters=3)
h = hierarchy.fit_predict(X)

plt.scatter(X[:, 0], X[:, 1], c=h, cmap='rainbow')
plt.title("Hierarchical Clustering")
plt.xlabel("Feature 1")
plt.ylabel("Feature 2")
plt.show()

print("K-Means Score:", silhouette_score(X, k))
print("Hierarchical Score:", silhouette_score(X, h))

# Implement and train a Convolutional Neural Network (CNN) for an imageclassification problem
# (e.g., handwritten digits, facial expressions, medical
# images, plant disease, or traffic signs) and evaluate model performance using
# appropriate metrics and validation techniques.

import tensorflow as tf
from sklearn.metrics import confusion_matrix, classification_report

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()

x_train = x_train / 255.0
x_test = x_test / 255.0

model = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(28, 28, 1)),
    tf.keras.layers.Conv2D(32, (3, 3), activation="relu"),
    tf.keras.layers.MaxPooling2D((2, 2)),
    tf.keras.layers.Flatten(),
    tf.keras.layers.Dense(64, activation="relu"),
    tf.keras.layers.Dense(10, activation="softmax")
])

model.compile(optimizer="adam",loss="sparse_categorical_crossentropy",metrics=["accuracy"])

model.fit(x_train, y_train, epochs=5,validation_split=0.1)

loss, accuracy = model.evaluate(x_test, y_test)
print("Test Accuracy:", accuracy)

pred = model.predict(x_test).argmax(axis=1)
print("Confusion Matrix:\n", confusion_matrix(y_test, pred))
print(classification_report(y_test, pred))
