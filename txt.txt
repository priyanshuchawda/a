import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import MinMaxScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.metrics import silhouette_score

data = load_iris()
X = data.data
X = MinMaxScaler().fit_transform(X)

kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
k = kmeans.fit_predict(X)

plt.scatter(X[:, 0], X[:, 1], c=k, cmap='viridis')
plt.title("K-Means Clustering")
plt.xlabel("Feature 1")
plt.ylabel("Feature 2")
plt.show()

hierarchy = AgglomerativeClustering(n_clusters=3)
h = hierarchy.fit_predict(X)

plt.scatter(X[:, 0], X[:, 1], c=h, cmap='rainbow')
plt.title("Hierarchical Clustering")
plt.xlabel("Feature 1")
plt.ylabel("Feature 2")
plt.show()

print("K-Means Score:", silhouette_score(X, k))
print("Hierarchical Score:", silhouette_score(X, h))


from sklearn.datasets import load_iris
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.neighbors import KNeighborsClassifier
from sklearn.model_selection import cross_val_score, KFold
from sklearn.pipeline import make_pipeline

# Load dataset
data = load_iris()
X = data.data
y = data.target

# Handle missing values
X = SimpleImputer(strategy="mean").fit_transform(X)

# Encode categorical target
y = LabelEncoder().fit_transform(y)

# 5-fold cross-validation
cv = KFold(n_splits=5, shuffle=True, random_state=42)

# Train and compare K-NN models
for k in [3, 5]:
    model = make_pipeline(
        StandardScaler(),
        KNeighborsClassifier(n_neighbors=k)
    )

    scores = cross_val_score(model, X, y, cv=cv)

    print("K =", k)
    print("Fold scores:", scores)
    print("Mean accuracy:", scores.mean())

import heapq

def heuristic(state, goal):
    distance = 0
    for i in range(9):
        if state[i] != 0:
            j = goal.index(state[i])
            distance += abs(i//3 - j//3) + abs(i%3 - j%3)
    return distance

def solve(start, goal):
    pq = [(heuristic(start, goal), 0, start, [])]
    visited = set()

    while pq:
        f, cost, state, path = heapq.heappop(pq)

        if state in visited:
            continue
        visited.add(state)
        path = path + [state]

        if state == goal:
            for s in path:
                print(s[:3])
                print(s[3:6])
                print(s[6:])
                print()
            print("Total cost:", cost)
            return

        z = state.index(0)
        for move in [-3, 3, -1, 1]:
            n = z + move
            if 0 <= n < 9 and not (
                move == -1 and z % 3 == 0 or
                move == 1 and z % 3 == 2
            ):
                new = state.copy()
                new[z], new[n] = new[n], new[z]
                if tuple(new) not in visited:
                    g = cost + 1
                    h = heuristic(new, goal)
                    heapq.heappush(
                        pq, (g + h, g, new, path)
                    )

from collections import deque

def moves(x,y,a,b):
	return [
		((a,y), "Fill Jug 1"),
		((x,b), "Fill Jug 2"),
		((0,y), "Empty Jug 1"),
		((x,0), "Empty Jug 2"),
		((x - min(x,b-y) , y+min(x,b-y)), "Pour 1 to 2"),
		((x+min(y,a-x) , y - min(y,a-x)) , "Pour 2 to 1")
	]

def bfs(a,b,goal):
	q = deque([((0,0) , [])])
	visited = set()

	while q:
		(x,y), path = q.popleft()

		if x == goal or y == goal:
			for action , state in path:
				print(action , "->" , state)
			print("goal reached:", (x,y))
			return

		if (x,y) in visited:
			continue
		visited.add((x,y))

		for state, action in moves (x,y,a,b):
			if state not in visited:
				q.append((state , path + [(action,state)]))
	print ("none")



def dfs(a, b, goal):
    stack = [((0, 0), [])]
    visited = set()

    while stack:
        (x, y), path = stack.pop()

        if x == goal or y == goal:
            for action, state in path:
                print(action, "->", state)
            print("Goal reached:", (x, y))
            return

        if (x, y) in visited:
            continue

        visited.add((x, y))

        for state, action in moves(x, y, a, b):
            if state not in visited:
                stack.append((state, path + [(action, state)]))

    print("None")
